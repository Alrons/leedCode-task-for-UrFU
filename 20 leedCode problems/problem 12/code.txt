/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public List<String> binaryTreePaths(TreeNode root) {
        List<String> res = new ArrayList<>();
        if (root != null) searchBT(root, new StringBuilder(), res);
        return res;

    }
    public void searchBT(TreeNode root, StringBuilder path, List<String> res){
        if (root == null) return;
    
        int len = path.length(); // Сохраняем текущую длину StringBuilder
    
        path.append(root.val);

        // Если узел - лист, добавляем путь в результат
        if (root.left == null && root.right == null) {
            res.add(path.toString());
        }
        else{
            path.append("->");
            searchBT(root.left, path, res);
            searchBT(root.right, path, res);
        }
    
        // Откатываем изменения (backtracking)
        path.setLength(len);
    }

}
