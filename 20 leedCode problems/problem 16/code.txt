class Solution {
    public List<String> generateParenthesis(int n) {
        List<String> res = new ArrayList<>();

        dfs(n, res, new StringBuilder(), 0, 0);
        return res;
    }

    public void dfs(int n, List<String> res, StringBuilder s, int openP, int closeP){
        if(openP == closeP && openP == n){
            res.add(s.toString());
        }

        if(openP < n){
            dfs(n, res, s.append('('), openP+1, closeP);
            s.deleteCharAt(s.length()-1);
        }
        if(openP > closeP){
            dfs(n, res, s.append(')'), openP, closeP+1);
            s.deleteCharAt(s.length()-1);
        }
    }
}
