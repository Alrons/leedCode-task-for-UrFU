class Solution {

    // разогрев JIT-компилятора, об этой фиче я узнал только сейчас. После нее результат получился на 2 мс быстрее
    static {
        Solution s = new Solution();
        for (int i = 0; i < 1000; ++i) {
            s.prefixesDivBy5(new int[] { 1, 0, 1 });
        }
    }

    public List<Boolean> prefixesDivBy5(int[] nums) {

        List<Boolean> result = new ArrayList<>();
        
        int remainder = 0; 

        for (int num : nums) {
            
            remainder = (remainder * 2 + num) % 5;
            
            if (remainder == 0) {
                result.add(true);
            } else {
                result.add(false);
            }
        }

        return result;
    }
}
