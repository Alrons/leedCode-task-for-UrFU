class Solution {
    public boolean isIsomorphic(String s, String t) {
        if(s.length() != t.length()){
            return false;
        }
        Map<Character, Character> mapChars = new HashMap<>(); 

        for(int i = 0; i < s.length(); i++){
            char charS = s.charAt(i);
            char charT = t.charAt(i);

            if(!mapChars.containsKey(charS)){
                if(mapChars.containsValue(charT)){
                    return false;
                }
                mapChars.put(charS, charT);
            }
            else if (mapChars.get(charS) != charT ){
                return false;
            }
            
        }
        return true;
    }
}
