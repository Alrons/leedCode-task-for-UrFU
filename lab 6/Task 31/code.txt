class Solution {
    public boolean wordPattern(String pattern, String s) {
        String[] splitS = s.split(" ");
        
        if (pattern.length() != splitS.length) {
            return false;
        }

        Map<Character, String> seen = new HashMap<>();

        for (int i = 0; i < pattern.length(); i++) {
            char currentChar = pattern.charAt(i);
            String currentWord = splitS[i];

            if (!seen.containsKey(currentChar)) {
                if (seen.containsValue(currentWord)) {
                    return false;
                }

                seen.put(currentChar, currentWord);
            } else {
                if (!seen.get(currentChar).equals(currentWord)) {
                    return false;
                }
            }
        }
        return true;
        
    }

}
