class Solution {
    public List<Integer> findDisappearedNumbers(int[] nums) {
        int length = nums.length;
        
        int[] seen = new int[length];
        for(int i = 0; i < length; i++){
            seen[nums[i]-1]++;
        }
        List<Integer> result = new ArrayList<>();

        for(int i = 0; i < length; i++){
            if(seen[i] == 0){
                result.add(i+1);
            }
        }
        return result;

    }
}
