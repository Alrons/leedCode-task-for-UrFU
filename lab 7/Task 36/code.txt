class Solution {
    public int climbStairs(int n) {
        
        if (n == 1){
            return 1;
        }
        if (n == 2){
            return 2;
        }

        int oneStap = 1;
        int twoStap = 2;

        int nexStap = 0;
        for (int i = 2; i < n; i++){
            nexStap = twoStap + oneStap;
            oneStap = twoStap;
            twoStap = nexStap;
        }
        return nexStap;
    }
}
