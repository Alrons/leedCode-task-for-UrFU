class Solution {
    public int smallestRepunitDivByK(int k) {
        
        if (k == 1){
            return 1;
        }
        if (k % 2 == 0 || k % 5 == 0){
            return -1;
        }

        int lenght = 1;
        int remainder = 1; 

        while(remainder != 0){
            remainder = (remainder * 10 + 1) % k;
            lenght++;
        }
        return lenght;

    }
}
