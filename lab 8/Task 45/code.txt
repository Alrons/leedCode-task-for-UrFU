class Solution {
    public int countNegatives(int[][] grid) {
        int result = 0;
        int rows = grid.length;
        int cols = grid[0].length; 

        int i = 0; // firstIndex (строка)
        int j = 0; // secondIndex (столбец)

        while (i < rows) { // Пока не прошли все строки
            
            // 1. Проверяем текущий элемент
            if (grid[i][j] < 0) {
                result++;
            }

            // 2. Двигаем индексы
            j++; 

            // 3. Если столбцы в этой строке кончились
            if (j == cols) {
                j = 0;
                i++; 
            }
        }

        return result;

    }
        
}
